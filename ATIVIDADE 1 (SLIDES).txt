ATIVIDADE 1 (SLIDES)

1. Listar todos os restaurantes:
db.restaurants.find({})

2. Encontrar todos os restaurantes do bairro (borough) "Manhattan":
db.restaurants.find({ borough: "Manhattan" })

3. Encontrar todos os restaurantes com CEP (zipcode) "10022":
db.restaurants.find({ "address.zipcode": "10022" })

4. Encontrar todos os restaurantes que possuam alguma nota (grade) "B":
db.restaurants.find({ "grades.grade": "B" })

5. Listar todos os restaurantes com pontuação (score) maior que 30:
db.restaurants.find({ "grades.score": { $gt: 30 } })

6. Listar todos os restaurantes com pontuação (score) menor que 10:
db.restaurants.find({ "grades.score": { $lt: 10 } })

7. Listar todos os restaurantes com culinária (cuisine) italiana ("Italian") E com CEP "10075":
db.restaurants.find({ cuisine: "Italian", "address.zipcode": "10075" })

8. Listar todos os restaurantes com culinária (cuisine) italiana ("Italian") OU com CEP "10075":
db.restaurants.find({
$or: [
{ cuisine: "Italian" },
{ "address.zipcode": "10075" }
]
})
